# Build the real Moonshine DwaBase shader with the same native AVX1 ABI.
project(Moonshine LANGUAGES CXX ISPC)
set(PACKAGE_NAME moonshine)
set(PROJECT_SOURCE_DIR "${ROOT}/upstream/openmoonray/moonray/moonshine")
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
include(MoonshineCompileDefinitions)
include(MoonshineCompileFeatures)
include(MoonshineLinkOptions)
include(MoonrayDso)
include(SConscriptStub)
function(Moonshine_cxx_compile_options target)
  target_compile_options(${target} PRIVATE -march=ivybridge -mno-avx2 -mno-fma -ffp-contract=off -fno-strict-aliasing)
endfunction()
function(Moonshine_ispc_compile_options target)
  get_target_property(output_dir ${target} BINARY_DIR)
  set_property(TARGET ${target} PROPERTY ISPC_HEADER_DIRECTORY "${output_dir}")
  target_compile_options(${target} PRIVATE ${GLOBAL_ISPC_FLAGS} --wno-perf --opt=disable-assertions)
  set_property(TARGET ${target} PROPERTY TARGET_OBJECTS $<TARGET_OBJECTS:${target}>)
endfunction()
file(COPY "${PROJECT_SOURCE_DIR}/lib/" DESTINATION "${PROJECT_BINARY_DIR}/include/moonshine"
     FILES_MATCHING PATTERN "*.h" PATTERN "*.isph")
include_directories("${PROJECT_BINARY_DIR}/include")
foreach(part common/interpolation map/projection material/glitter material/dwabase)
  add_subdirectory("${PROJECT_SOURCE_DIR}/lib/${part}" "lib/${part}")
endforeach()
foreach(target common_interpolation map_projection material_glitter material_dwabase)
  target_link_libraries(${target} PRIVATE ispc_runtime)
  target_include_directories(${target}_objlib PRIVATE "${PROJECT_BINARY_DIR}/include")
  get_target_property(output_dir ${target}_objlib BINARY_DIR)
  file(RELATIVE_PATH part "${PROJECT_BINARY_DIR}/lib" "${output_dir}")
  file(MAKE_DIRECTORY "${PROJECT_BINARY_DIR}/include/moonshine/${part}")
  get_target_property(ispc_sources ${target}_objlib SOURCES)
  set(headers)
  foreach(source IN LISTS ispc_sources)
    if(source MATCHES "\\.ispc$")
      get_filename_component(stem "${source}" NAME_WE)
      set(header "${PROJECT_BINARY_DIR}/include/moonshine/${part}/${stem}_ispc_stubs.h")
      add_custom_command(OUTPUT "${header}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${output_dir}/${stem}_ispc_stubs.h" "${header}"
        DEPENDS ${target}_objlib "${output_dir}/${stem}_ispc_stubs.h")
      list(APPEND headers "${header}")
    endif()
  endforeach()
  add_custom_target(${target}_headers DEPENDS ${headers})
  add_dependencies(${target} ${target}_headers)
endforeach()
# Compile every upstream MoonShine material, including compound and hair shaders.
add_subdirectory("${PROJECT_SOURCE_DIR}/dso/material" materials)
set(material_targets DwaAdjustMaterial DwaBaseMaterial DwaColorCorrectMaterial
  DwaEmissiveMaterial DwaFabricMaterial DwaLayerMaterial DwaMetalMaterial
  DwaMixMaterial DwaRefractiveMaterial DwaSkinMaterial DwaSolidDielectricMaterial
  DwaSwitchMaterial DwaTwoSidedMaterial DwaVelvetMaterial_v2 HairMaterial_v3
  HairColorCorrectMaterial HairDiffuseMaterial HairLayerMaterial DwaToonMaterial HairToonMaterial)
set(all_material_targets)
foreach(target IN LISTS material_targets)
  target_link_libraries(${target} PRIVATE ispc_runtime)
  target_link_libraries(${target}_proxy PRIVATE ispc_runtime)
  list(APPEND all_material_targets ${target} ${target}_proxy)
endforeach()
add_custom_target(moonshine_material_library DEPENDS ${all_material_targets})

# Texture and normal shaders share the same AVX ABI as the material library.
add_subdirectory("${PROJECT_SOURCE_DIR}/dso/map" maps)
add_subdirectory("${PROJECT_SOURCE_DIR}/dso/normalmap" normalmaps)
collect_desktop_dsos("${PROJECT_SOURCE_DIR}/dso/map" map_targets)
collect_desktop_dsos("${PROJECT_SOURCE_DIR}/dso/normalmap" normalmap_targets)
add_custom_target(moonshine_map_library DEPENDS ${map_targets} ${normalmap_targets})
add_dependencies(moonshine_material_library moonshine_map_library)

# Example scenes use these analytic geometry DSOs as well as core mesh geometry.
add_subdirectory("${PROJECT_SOURCE_DIR}/dso/geometry" geometries)
add_custom_target(moonshine_geometry_library DEPENDS SphereGeometry SphereGeometry_proxy BoxGeometry BoxGeometry_proxy)
add_dependencies(moonshine_material_library moonshine_geometry_library)
