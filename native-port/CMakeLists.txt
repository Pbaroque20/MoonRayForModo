function(desktop_exclude_sources target)
  get_target_property(source_dir ${target} SOURCE_DIR)
  get_target_property(sources ${target} SOURCES)
  set(kept)
  foreach(source IN LISTS sources)
    get_filename_component(name "${source}" NAME)
    if(NOT name IN_LIST ARGN)
      if(IS_ABSOLUTE "${source}" OR source MATCHES "^\\$<")
        list(APPEND kept "${source}")
      else()
        list(APPEND kept "${source_dir}/${source}")
      endif()
    endif()
  endforeach()
  set_property(TARGET ${target} PROPERTY SOURCES "${kept}")
endfunction()
